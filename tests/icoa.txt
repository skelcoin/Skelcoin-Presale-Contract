describe("ICO Program Tests", () => {
  const provider = anchor.AnchorProvider.env();
  anchor.setProvider(provider);
  // @ts-ignore
  const program = anchor.workspace.Ico as Program<Ico>;
  const connection = provider.connection;

it("Fetches feed value", async () => {
    // Example: SOL/USD feed on devnet (replace with your feed if using localnet)
    const FEED_PUBKEY = new anchor.web3.PublicKey(
      "DAXAq94Y5nX2dDp15SdeBzYRqTn8viFf9Dxq4ws7rHec" // sample Switchboard feed

    try {
      const tx = await program.methods

      console.log("✅ Transaction success:", tx);

      console.error("❌ Error calling solana_price:", err);
